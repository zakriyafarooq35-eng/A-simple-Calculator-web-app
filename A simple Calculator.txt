<script>

let current = "";
let previous = "";
let operator = "";

const display = document.getElementById("display");
const previousDisplay = document.getElementById("previous");


function updateDisplay() {
    display.value = current || "0";
}


function number(value) {

    if (current === "0") {
        current = value;
    } else {
        current += value;
    }

    updateDisplay();
}


function decimal() {

    if (!current.includes(".")) {

        current = current || "0";
        current += ".";

        updateDisplay();
    }
}


function selectOperator(op) {

    if (current === "") return;

    if (previous !== "") {
        calculate();
    }

    previous = current;
    current = "";
    operator = op;

    previousDisplay.textContent =
        previous + " " + symbol(op);
}


function symbol(op) {

    if (op === "*") return "×";
    if (op === "/") return "÷";
    if (op === "-") return "−";
    if (op === "+") return "+";

    return "";
}


function calculate() {

    if (
        previous === "" ||
        current === "" ||
        operator === ""
    ) {
        return;
    }

    let a = parseFloat(previous);
    let b = parseFloat(current);
    let result;


    if (operator === "+") {
        result = a + b;
    }

    else if (operator === "-") {
        result = a - b;
    }

    else if (operator === "*") {
        result = a * b;
    }

    else if (operator === "/") {

        if (b === 0) {

            display.value = "Error";

            current = "";
            previous = "";
            operator = "";

            previousDisplay.textContent = "";

            return;
        }

        result = a / b;
    }


    result = parseFloat(result.toFixed(10));

    previousDisplay.textContent =
        a + " " + symbol(operator) + " " + b + " =";

    current = String(result);
    previous = "";
    operator = "";

    updateDisplay();
}


function clearCalculator() {

    current = "";
    previous = "";
    operator = "";

    previousDisplay.textContent = "";

    updateDisplay();
}


function deleteNumber() {

    current = current.slice(0, -1);

    updateDisplay();
}


function percentage() {

    if (current !== "") {

        current =
            String(parseFloat(current) / 100);

        updateDisplay();
    }
}


/* Keyboard Support */

document.addEventListener("keydown", function(event) {

    if (event.key >= "0" && event.key <= "9") {
        number(event.key);
    }

    else if (event.key === ".") {
        decimal();
    }

    else if (event.key === "+") {
        selectOperator("+");
    }

    else if (event.key === "-") {
        selectOperator("-");
    }

    else if (event.key === "*") {
        selectOperator("*");
    }

    else if (event.key === "/") {
        event.preventDefault();
        selectOperator("/");
    }

    else if (event.key === "Enter" || event.key === "=") {
        calculate();
    }

    else if (event.key === "Backspace") {
        deleteNumber();
    }

    else if (event.key === "Escape") {
        clearCalculator();
    }

    else if (event.key === "%") {
        percentage();
    }

});
